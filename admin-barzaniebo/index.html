ڕێکخستنی بنکەی دراوەی Supabase SQL

ئەم کۆدەی خوارەوە بە ١ کرتە کۆپی بکە و لە Supabase SQL Editor پیستی بکە و RUN داگرە تا سڕینەوە و ئاگادارییەکان ١٠٠٪ چالاک ببن.

کردنەوەی SQL Editor
-- ============================================================== -- کۆدی ڕێکخستنی تەواوی خشتەکانی کۆماندۆ لە Supabase -- ئەم کۆدە لە Supabase -> SQL Editor کۆپی بکە و RUN داگرە -- ============================================================== -- ١. سڕینەوەی کاڵا و ئۆفەرە نموونەییە کۆنەکان بۆ ئەوەی بنکەی دراوە تەواو بەتاڵ بێت TRUNCATE TABLE products CASCADE; TRUNCATE TABLE offers CASCADE; -- ٢. ڕێگەدان بە دەشبۆرد و ئەپ بۆ سڕینەوە، زیادکردن، و دەستکاریکردن بەبێ ڕێگری (Disable RLS) ALTER TABLE IF EXISTS products DISABLE ROW LEVEL SECURITY; ALTER TABLE IF EXISTS offers DISABLE ROW LEVEL SECURITY; ALTER TABLE IF EXISTS orders DISABLE ROW LEVEL SECURITY; -- ٣. دروستکردنی خشتەی ئاگادارییەکان بۆ ئەپ CREATE TABLE IF NOT EXISTS notifications ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), title text NOT NULL, body text NOT NULL, category text DEFAULT 'ئاگاداری', time text DEFAULT '', is_unread boolean DEFAULT true, created_at timestamptz DEFAULT now() ); ALTER TABLE notifications DISABLE ROW LEVEL SECURITY; -- ٤. دروستکردنی خشتەی بانەرە جوڵاوەکانی ئەپ CREATE TABLE IF NOT EXISTS banners ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), image_url text NOT NULL, title text DEFAULT '', link_url text DEFAULT '', created_at timestamptz DEFAULT now() ); ALTER TABLE banners DISABLE ROW LEVEL SECURITY; -- ٥. دڵنیابوونەوە لە هەبوونی خشتەی داواکارییەکان (Orders) CREATE TABLE IF NOT EXISTS orders ( id text PRIMARY KEY, customer text NOT NULL, phone text DEFAULT '', address text DEFAULT '', items text DEFAULT '', total numeric DEFAULT 0, status text DEFAULT 'pending', created_at timestamptz DEFAULT now() ); ALTER TABLE IF EXISTS orders DISABLE ROW LEVEL SECURITY; -- ٦. ڕێگەدان بە دەشبۆرد و ئەپ بۆ پیشاندانی بەکارهێنەرانی چوونەژوورەوە (Profiles / Users) CREATE TABLE IF NOT EXISTS profiles ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), full_name text, phone text, avatar_url text, updated_at timestamptz DEFAULT now() ); ALTER TABLE IF EXISTS profiles DISABLE ROW LEVEL SECURITY;
